\section{Tuplas}

\begin{frame}[fragile]{Tuplas}
    \begin{itemize}
        \item Secuencia {\blue ordenada} de elementos, que pueden ser de diferentes {\blue tipos}.
        \item No es posible alterar sus elementos, son {\bfseries\orange inmutables}.
        \item Se representan con paréntesis
    \end{itemize}
\begin{lstlisting}
# Tupla vacia
te = ()
# Tupla de ejemplo
t = (2, "pes", 3.2)

t[0]      # devuelve 2
t + (5,6) # devuelve (2, "pes" 3.2, 5, 6)
t[1:2]    # slice, devuelve ("pes", )
t[1:3]    # slice, devuelve ("pes", 3.2)
len(t)    # devuelve 3

t[1] = 4 # error, inmutable
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Iteración sobre tuplas}
    \begin{itemize}
        \item Es posible {\blue iterar} sobre los elementos de una tupla:
    \end{itemize}
\begin{lstlisting}
t = (2, "pes", 3.2)

for elem in t: 
    print(elem, "es de tipo", type(elem))
\end{lstlisting}

% Iteration over tuples with pairs
\begin{frame}[fragile]{Tuplas de tuplas}
\begin{lstlisting}
def get_data(aTuple):
    nums= ()
    words = ()
    for t in aTuple:
        nums= nums+ (t[0],)
        if t[1] not in words:
            words = words + (t[1],)
    
    min_n = min(nums)
    max_n = max(nums)
    unique_words = len(words)
    return (min_n, max_n, unique_words)
    
aTuple = ((1, 'a'), (2, 'b'), (3, 'c'))
print(get_data(aTuple))
\end{lstlisting}
\end{frame}

\section{Listas}
\begin{frame}[fragile]{Listas}
    \begin{itemize}
        \item {\blue\bfseries Secuencia ordenada} de elementos, accesibles a través de un {\blue índice}.
        \item Se denotan utilizando {\blue corchetes}, \lstinline|[ ]|
        \item Los elementos: 
            \begin{itemize}
                \item Son usualmente {\orange homogéneos}
                \item Pero pueden ser de diferentes tipos (práctica poco común, pero válida).
            \end{itemize}
        
        \item Los elementos pueden ser {\orange\bfseries alterados}, por lo que la lista es {\orange\bfseries mutable}.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Lista e índices}
\begin{lstlisting}
L = []
L = [2, 'a', 4, [1,2]]

len(L) # evaluates to 4
L[0]   # evaluates to 2
L[2]+1 # evaluates to 5
L[3]   # evaluates to [1,2], another list!
L[4]   # gives an error

i = 2
L[i-1] # evaluates to 'a' since L[1]='a' above
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Otras operaciones sobre listas}
    \begin{itemize}
        \item Funciones para ordenar: \lstinline|sort()| y \lstinline|sorted()|
        \item Operación de {\blue reversa}: \lstinline|reverse()|
        \item Otras operaciones más: \url{https://docs.python.org/3/tutorial/datastructures.html}

\end{lstlisting}

\section{Mutabilidad y clonado}

\begin{frame}[fragile]{Listas en memoria}
    \begin{itemize}
        \item Las listas son objetos {\blue mutables}.
        \item Se comportan de forma diferente a los tipos inmutables.
        \item Representan {\orange objetos en memoria}. 
        \item Distintos {\orange nombres de variables} pueden apuntar al mismo objeto en memoria.
        \item Cualquier variable que {\blue\bfseries apunte } a un objeto es afectada.
    \end{itemize}
    
    \bottomline{Cuando trabajemos con lista debemos cuidarnos de los efectos secundarios.}
\end{frame}

\begin{frame}[fragile]{Alias para listas}
    \begin{itemize}
        \item \lstinline|hot| es un alias para \lstinline|warm|: al cambiar uno, cambia el otro.
        
        \item La función \lstinline|append()| tiene un {\blue efecto secundario.}

\begin{lstlisting}
a = 1
b = a
print(a)
print(b)

warm = ['red', 'yellow', 'orange']
hot = warm
hot.append('pink')
print(hot)
print(warm)
\end{lstlisting}

% Mutación e iteración - evitar mutar una lista al iterar sobre ella
\begin{frame}[fragile]{}
    \begin{itemize}
        \item Como {\blue\bfseries buena práctica}, es mejor evitar mutar una lista al iterar sobre ella. Veamos un ejemplo: 
        
        \item \href{http://pythontutor.com/visualize.html#code=def%20remove_dups%28L1,%20L2%29%3A%0A%20%20%20%20for%20e%20in%20L1%3A%0A%20%20%20%20%20%20%20%20if%20e%20in%20L2%3A%0A%20%20%20%20%20%20%20%20%20%20%20%20L1.remove%28e%29%0A%0AL1%20%3D%20%5B1,%202,%203,%204%5D%0AL2%20%3D%20%5B1,%202,%205,%206%5D%0Aremove_dups%28L1,%20L2%29%0Aprint%28L1,%20L2%29&cumulative=false&curInstr=15&heapPrimitives=nevernest&mode=display&origin=opt-frontend.js&py=3&rawInputLstJSON=%5B%5D&textReferences=false}{Ver en Python Tutor}
        
        \item ¿Por qué \lstinline|L1 = [2,3,4]| y no \lstinline|[3,4]|? ¿Cómo se puede corregir este código?
    \end{itemize}
\begin{lstlisting}
def remove_dups(L1, L2):
    for e in L1:
        if e in L2:
            L1.remove(e)

L1 = [1, 2, 3, 4]
L2 = [1, 2, 5, 6]
remove_dups(L1, L2)
print(L1, L2)
\end{lstlisting}
\end{frame}

% Corrección
\begin{frame}[fragile]{}
    \begin{itemize}
        \item Solución.
    \end{itemize}
\begin{lstlisting}
def remove_dups_new(L1, L2):
    L1_copy = L1[:]
    for e in L1_copy:
        if e in L2:
            L1.remove(e)
            
L1 = [1, 2, 3, 4]
L2 = [1, 2, 5, 6]
remove_dups_new(L1, L2)
print(L1, L2)

\end{lstlisting}
\end{frame}
