\begin{frame}[fragile]{}
    \begin{itemize}
        \item {\blue\bfseries Programación defensiva}
        \begin{itemize}
            \item Escribir {\orange especificaciones} para las funciones.
            \item {\orange Modularizar} los programas.
            \item Revisar {\orange condiciones} sobre entradas y salidas (\textit{assertions}).
        \end{itemize}
        
        \item {\blue\bfseries Validación y pruebas}
        \begin{itemize}
            \item {\orange Comparar} pares de entradas y salidas para verificar las especificaciones.
            \item ¡No está funcionando!
            \item Pensar: ¿cómo puedo hacer que falle mi programa?
        \end{itemize}
        
        \item {\blue\bfseries Depuración (\textit{debugging})}
        \begin{itemize}
            \item {\orange Estudiar eventos} que han llevado a errores.
            \item Pensar: ¿por qué no está funcionando?
            \item ¿Cómo puedo arreglar mi programa?
        \end{itemize}
    \end{itemize}
\end{frame}

% Set up for easy testing
\begin{frame}[fragile]{Programación defensiva}
    \begin{itemize}
        \item Desde el {\orange\bfseries principio}, {\blue\bfseries diseñar} el código para facilitar esta parte.
        
        \item Diseñar el programa en {\blue módulos} que puedan ser probados y depurados individualmente.
        
        \item {\blue\bfseries Documentar las restricciones} sobre módulos:
        \begin{itemize}
            \item ¿Cuál se espera que sea la entrada?
            \item ¿Cuál se espera que sea la salida?
        \end{itemize}
        
        \item {\blue\bfseries Documentar supuestos} detrás del código.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{¿Cuándo se debe probar un código?}
    \begin{itemize}
        \item Asegurarse que el código {\blue\bfseries pueda ejecutarse}.
        \begin{itemize}
            \item Eliminar errores de sintaxis, semántica estática.
            \item El intérprete de Python y los editores ayudan en esta tarea.
        \end{itemize}
        
        \item Contar con un {\blue\bfseries conjunto de prueba}.
        \begin{itemize}
            \item Conjunto de entrada del cual conocemos la salida esperada.
        \end{itemize}
    \end{itemize}
\end{frame}

% Tipos de pruebas
\begin{frame}[fragile]{Clases de pruebas}
    \begin{itemize}
        \item {\blue\bfseries Pruebas de unidad}
        \begin{itemize}
            \item Validar cada pieza de un programa.
            \item {\orange Probar cada función} de forma separada.
        \end{itemize}
        
        \item {\blue\bfseries Pruebas de regresión}
        \begin{itemize}
            \item Añadir casos de pruebas conforme se encuentran \textit{bugs}.
            \item {\orange Validar} errores reintroducidos y previamente arreglados.
        \end{itemize}
        
        \item {\blue\bfseries Pruebas de integración}
        \begin{itemize}
            \item ¿Cómo unir todo el programa?, ¿funciona en conjunto?
            \item No se debe apresurar en este paso.
        \end{itemize}
    \end{itemize}
\end{frame}


% Tipos de pruebas
\begin{frame}[fragile]{Tipos de pruebas}
    \begin{itemize}
        \item {\blue\bfseries Intuición} acerca de los límites del programa.
\begin{lstlisting}
def is_bigger(x, y):
    ''' Assumes x and y are ints
    Returns True if y is less than x, else False '''
\end{lstlisting}
        \begin{itemize}
            \item ¿alguna partición natural para probar esta función?
        \end{itemize}
        
        \item Si no hay particiones naturales $\Rightarrow$ {\blue pruebas aleatorias}.
        \begin{itemize}
            \item Más pruebas $\Rightarrow$ mayor prob. de éxito. 
            \item Método no ideal.
        \end{itemize}
        
        \item {\orange\bfseries Prueba de caja negra}
        \begin{itemize}
            \item Explorar casos a través de la especificación.
        \end{itemize}
        
        \item {\orange\bfseries Prueba de caja de cristal}
        \begin{itemize}
            \item Explorar casos a través del conocimiento del código.
        \end{itemize}
    \end{itemize}
\end{frame}

% Pruebas de caja negra
\begin{frame}[fragile]{Pruebas de caja negra}
\begin{lstlisting}
def sqrt(x, eps):
    ''' Assumes x, eps floats, x >= 0, eps > 0
    Returns res such that x-eps <= res*res <= x+eps '''
\end{lstlisting}
    \begin{itemize}
        \item Diseñada {\orange sin conocimiento} del código.
        \item Puede ser realizada por alguien distinto al programador.
        \item Puede ser {\orange reutilizada} aún si la implementación cambia.
        \item Algunos caminos:
        \begin{itemize}
            \item Casos de prueba en particiones naturales.
            \item Considera {\orange condiciones límite}: listas vacías, listas \textit{singleton}, números muy grandes o muy pequeños, etc.
        \end{itemize}
    \end{itemize}
\end{frame}

% Pruebas de caja de cristal
\begin{frame}[fragile]{Pruebas de caja de cristal}
    \begin{itemize}
        \item Utilizamos {\orange el código} para construir los casos de prueba.
        \item Pueden ser {\orange completas} si se explora cada ramificación del código al menos una vez.
        \item Desventajas:
        \begin{itemize}
            \item Muchas iteraciones.
            \item Nos podemos saltar alguna rama.
        \end{itemize}
        
        \item Guía para diseñarlas:
        \begin{itemize}
            \item Revisar condicionales: ¿están todos los posibles casos?
            \item Revisar ciclos: ¿entra al ciclo?, ¿por qué se ejecuta una vez, o más de una vez?
            \item Tipos de datos: ¿Devuelve el tipo esperado?, ¿se tuvo cuidado al operar las variables?
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{}
\begin{lstlisting}
def abs(x):
    """ Assumes x is an int
    Returns x if x>=0 and -x otherwise """
    if x < -1:
        return -x
    else:
        return x
\end{lstlisting}
    \begin{itemize}
        \item Aún si la prueba es {\orange completa}, podríamos saltarnos un \textit{bug}.
        \item ¿Qué pasa con los casos 2 y -2?
        \item ¿Qué pasa con \lstinline|abs(-1)|?
        \item Se deben probar casos {\orange\bfseries límite}.
    \end{itemize}
\end{frame}

% Manejo de excepciones
\begin{frame}[fragile]{Manejo de excepciones}
    \begin{itemize}
        \item Python provee bloques de {\blue\bfseries manejo} de excepciones (exceptions).
\begin{lstlisting}
try:
    a = int(input("Tell me one number:"))
    b = int(input("Tell me another number:"))
    print(a/b)
except:
    print("Bug in user input.")
\end{lstlisting}

        \item Las {\orange\bfseries excepciones} que surjan (\textit{raised by}) en el cuerpo de una sentencia \lstinline|try| son manejadas por el cuerpo de \lstinline|except| mientras {\blue la ejecución continúa}.
    \end{itemize}
\end{frame}

% Excepciones específicas
\begin{frame}[fragile]{Excepciones específicas}
    \begin{itemize}
        \item Es posible especificar cláusulas de \lstinline|except| para una excepción en particular. 
\begin{lstlisting}
try:
    a = int(input("Tell me one number: "))
    b = int(input("Tell me another number: "))
    print("a/b = ", a/b)
    print("a+b = ", a+b)
except ValueError:
    print("Could not convert to a number.")
except ZeroDivisionError:
    print("Can't divide by zero")
except:
    print("Something went very wrong.")
\end{lstlisting}

    \end{itemize}
\end{frame}

% Más cláusulas
\begin{frame}[fragile]{Otras excepciones}
    \begin{itemize}
        \item \lstinline|else|:
        \begin{itemize}
            \item Se ejecuta si la ejecución de \lstinline|try| {\orange se completa sin excepciones}.
        \end{itemize}
        
        \item \lstinline|finally|:
        \begin{itemize}
            \item Se ejecuta {\orange\bfseries siempre} después de \lstinline|try|, \lstinline|else| y \lstinline|except| o si ejecuta \lstinline|break|, \lstinline|continue| o \lstinline|return|.
            
            \item {\orange\bfseries Útil} para limpiar el código con instrucciones que de todas formas deben ejecutarse sin importar el resultado, como cerrar un archivo, conexión HTTP, etc.
        \end{itemize}
    \end{itemize}
\end{frame}

% ¿Qué hacer con las excepciones?
\begin{frame}[fragile]{¿Qué hacer con las excepciones?}
    \begin{itemize}
        \item {\blue\bfseries Manejarlas silenciosamente}
        \begin{itemize}
            \item Sustituir valores por defecto, solamente continuar con el código.
            \item ¡Mala idea! el usuario no tiene advertencias.
        \end{itemize}
        
        \item Devolver un valor para {\blue\bfseries representar} el error. 
        \begin{itemize}
            \item ¿Qué valor escoger?
            \item Complica el resto del código.
        \end{itemize}
        
        \item Detener la ejecución, {\orange\bfseries mostrar una condición de error}:
        \begin{itemize}
            \item En Python: \lstinline|raise|
        \end{itemize}
\begin{lstlisting}
raise Exception("string descriptivo")
\end{lstlisting}
    \end{itemize}
\end{frame}

% Exception example
\begin{frame}[fragile]{Excepciones: ejemplo}
\begin{lstlisting}
def get_ratios(L1, L2):
    """ Assumes: L1, L2 lists of equal length of numbers
        Returns: a list containing L1[i]/L2[i] """
    ratios = []
    for index in range(len(L1)):
        try:
            ratios.append(L1[index]/L2[index])
        except ZeroDivisionError:
            ratios.append(float('nan'))
            #nan = not a number
        except:
            raise ValueError('called with bad arg')
    return ratios
\end{lstlisting}
\end{frame}

\subsection{Assertions}

\begin{frame}[fragile]{Uso de \lstinline|assert|}
    \begin{itemize}
        \item Cuando quieres estar {\blue\bfseries seguro} de que los supuestos del cómputo son como se esperaba.
        
        \item Utilizamos una cláusula \lstinline|assert| para generar una excepción \lstinline|AssertionError| si no se verifican los supuestos.
        
        \item Este es un ejemplo de buena {\blue\bfseries programación defensiva}.
    \end{itemize}
\end{frame}

% Ejemplo de assert
\begin{frame}[fragile]{Ejemplo}
\begin{lstlisting}
def avg(grades):
    """ grades : grades list """
    assert len(grades) != 0, 'no grades data'
    return sum(grades)/len(grades)
\end{lstlisting}

    \begin{itemize}
        \item Esto lleva a una excepción de tipo \lstinline|AssertionError| si se da una lista vacía de notas.
        
        \item De lo contrario, ¡OK!
    \end{itemize}
\end{frame}

% Assertions como programación defensiva
\begin{frame}[fragile]{\textit{Assertions} como programación defensiva}
    \begin{itemize}
        \item Las afirmaciones no permiten que el programador controle respuesta a condiciones inesperadas.
        
        \item Garantizan que la {\blue ejecución se detenga} siempre que condición no se cumple.
        
        \item Normalmente se usan para {\blue verificar las entradas} a las funciones, pero pueden ser usadas en cualquier lugar.
        
        \item Pueden usarse para {\blue verificar las salidas} de una función para evitar propagar malos valores.
        
        \item Pueden facilitar la depuración de un error.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{¿Dónde usarlas?}
    \begin{itemize}
        \item El objetivo es encontrar errores tan pronto como se introduzcan.
        
        \item {\blue\bfseries Suplemento} a la etapa de pruebas.
        
        \item Levantar {\blue\bfseries excepciones} si el usuario provee datos incorrectos.
        
        \item También permiten: 
        \begin{itemize}
            \item Verificar {\blue\bfseries tipos} de los argumentos o valores.
            \item Revisar {\blue\bfseries anomalías} en las estructuras de datos.
            \item Revisar {\blue\bfseries restricciones} en los valores de retorno.
            \item Revisar {\blue\bfseries violaciones} en las restricciones de un procedimiento.
        \end{itemize}
    \end{itemize}
\end{frame}
